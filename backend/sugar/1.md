### wepay

```sql
CREATE TABLE `WOrder` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `out_trade_no` VARCHAR(64) NOT NULL COMMENT '商户订单号',
    `order_status` VARCHAR(20) NOT NULL DEFAULT 'UNPAID' COMMENT '订单状态',
    `total_amount` INT NOT NULL DEFAULT 0 COMMENT '支付金额（单位：分）',
    `open_id` VARCHAR(64) NOT NULL COMMENT '用户openid',
    `transaction_id` VARCHAR(64) DEFAULT NULL COMMENT '微信支付订单号',
    `updated_at` BIGINT DEFAULT NULL COMMENT '更新时间戳（毫秒）',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_out_trade_no` (`out_trade_no`),
    INDEX `idx_open_id` (`open_id`),
    INDEX `idx_transaction_id` (`transaction_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='支付订单表';
```
